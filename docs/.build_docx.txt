function W([string]$s) { [void]$script:doc.Append($s) }

function XmlEsc([string]$s) {
    $s = $s -replace '&', '&amp;'
    $s = $s -replace '<', '&lt;'
    $s = $s -replace '>', '&gt;'
    $s = $s -replace '"', '&quot;'
    return $s
}

function RunsCode([string]$text, [string]$rpr) {
    $out = ''
    $bt = [char]0x60
    $parts = [regex]::Split($text, '(' + $bt + '[^' + $bt + ']+' + $bt + ')')
    foreach ($p in $parts) {
        if ([string]::IsNullOrEmpty($p)) { continue }
        if ($p.Length -gt 2 -and $p[0] -eq $bt -and $p[$p.Length - 1] -eq $bt) {
            $t = XmlEsc $p.Substring(1, $p.Length - 2)
            $out += '<w:r><w:rPr>' + $rpr + '<w:rFonts w:ascii="Consolas" w:hAnsi="Consolas"/><w:color w:val="A02020"/><w:sz w:val="18"/><w:shd w:val="clear" w:fill="F2F2F2"/></w:rPr><w:t xml:space="preserve">' + $t + '</w:t></w:r>'
        } else {
            $t = XmlEsc $p
            $r = ''
            if ($rpr -ne '') { $r = '<w:rPr>' + $rpr + '</w:rPr>' }
            $out += '<w:r>' + $r + '<w:t xml:space="preserve">' + $t + '</w:t></w:r>'
        }
    }
    return $out
}

function Runs([string]$text, [string]$baseRpr) {
    $out = ''
    $parts = [regex]::Split($text, '(\*\*.+?\*\*)')
    foreach ($p in $parts) {
        if ([string]::IsNullOrEmpty($p)) { continue }
        if ($p.Length -gt 4 -and $p.StartsWith('**') -and $p.EndsWith('**')) {
            $inner = $p.Substring(2, $p.Length - 4)
            $out += RunsCode $inner ($baseRpr + '<w:b/>')
        } else {
            $out += RunsCode $p $baseRpr
        }
    }
    if ($out -eq '') { $out = '<w:r><w:t/></w:r>' }
    return $out
}

function Para([string]$style, [string]$text, [string]$baseRpr) {
    W ('<w:p><w:pPr><w:pStyle w:val="' + $style + '"/></w:pPr>' + (Runs $text $baseRpr) + '</w:p>')
}

function Flush-Table {
    if ($script:tblBuf.Count -eq 0) { return }
    $rows = @()
    foreach ($r in $script:tblBuf) {
        if ($r -match '^\s*\|[\s:\-\|]+\|\s*$') { continue }
        $cells = $r.Trim().Trim('|') -split '\|'
        $rows += , @($cells | ForEach-Object { $_.Trim() })
    }
    if ($rows.Count -eq 0) { $script:tblBuf.Clear(); return }

    if ($rows.Count -gt 1 -and $rows[0].Count -eq 5 -and $rows[0][0] -eq '#' -and $rows[0][4] -eq ([string][char]0xD6 + 'ncelik')) {
        for ($k = 1; $k -lt $rows.Count; $k++) {
            $r2 = @([string]$script:curPart) + $rows[$k]
            [void]$script:masterImg.Add($r2)
        }
    }

    $nCol = ($rows | ForEach-Object { $_.Count } | Measure-Object -Maximum).Maximum
    $colW = [int](9350 / $nCol)

    W '<w:tbl><w:tblPr><w:tblW w:w="5000" w:type="pct"/>'
    TblBorders
    W '</w:tblPr><w:tblGrid>'
    for ($c = 0; $c -lt $nCol; $c++) { W ('<w:gridCol w:w="' + $colW + '"/>') }
    W '</w:tblGrid>'

    for ($i = 0; $i -lt $rows.Count; $i++) {
        $isHead = ($i -eq 0)
        W '<w:tr>'
        if ($isHead) { W '<w:trPr><w:tblHeader/></w:trPr>' }
        for ($c = 0; $c -lt $nCol; $c++) {
            $val = ''
            if ($c -lt $rows[$i].Count) { $val = $rows[$i][$c] }
            W ('<w:tc><w:tcPr><w:tcW w:w="' + $colW + '" w:type="dxa"/>')
            if ($isHead) { W '<w:shd w:val="clear" w:fill="DCE6F1"/>' }
            W '</w:tcPr>'
            $rpr = '<w:sz w:val="18"/>'
            if ($isHead) { $rpr = '<w:b/><w:sz w:val="18"/>' }
            W ('<w:p><w:pPr><w:pStyle w:val="TableCell"/></w:pPr>' + (Runs $val $rpr) + '</w:p>')
            W '</w:tc>'
        }
        W '</w:tr>'
    }
    W '</w:tbl>'
    W '<w:p><w:pPr><w:pStyle w:val="AfterTable"/></w:pPr></w:p>'
    $script:tblBuf.Clear()
}

function Flush-Quote {
    if ($script:quoteBuf.Count -eq 0) { return }
    $joined = $script:quoteBuf -join ' '
    $n = $script:quoteBuf.Count
    for ($i = 0; $i -lt $n; $i++) {
        $st = 'Callout'
        if ($i -eq 0)        { $st = 'CalloutFirst' }
        if ($i -eq ($n - 1)) { $st = 'CalloutLast' }
        if ($n -eq 1)        { $st = 'CalloutOnly' }
        Para $st $script:quoteBuf[$i] '<w:sz w:val="18"/>'
    }
    if ($joined -match ([char]0x47 + [char]0xD6 + 'RSEL')) {
        $script:imgCount++
        W '<w:tbl><w:tblPr><w:tblW w:w="5000" w:type="pct"/>'
        W '<w:tblBorders><w:top w:val="dashed" w:sz="6" w:color="B8B8B8"/><w:left w:val="dashed" w:sz="6" w:color="B8B8B8"/><w:bottom w:val="dashed" w:sz="6" w:color="B8B8B8"/><w:right w:val="dashed" w:sz="6" w:color="B8B8B8"/></w:tblBorders>'
        W '</w:tblPr><w:tblGrid><w:gridCol w:w="9350"/></w:tblGrid>'
        W '<w:tr><w:trPr><w:trHeight w:val="3400" w:hRule="atLeast"/></w:trPr>'
        W '<w:tc><w:tcPr><w:tcW w:w="9350" w:type="dxa"/><w:shd w:val="clear" w:fill="FCFCFC"/><w:vAlign w:val="center"/></w:tcPr>'
        W ('<w:p><w:pPr><w:pStyle w:val="ImgBox"/></w:pPr><w:r><w:rPr><w:i/><w:color w:val="B0B0B0"/><w:sz w:val="18"/></w:rPr><w:t>' + (XmlEsc ('g' + [char]0xF6 + 'rsel buraya eklenecek')) + '</w:t></w:r></w:p>')
        W '</w:tc></w:tr></w:tbl>'
        W '<w:p><w:pPr><w:pStyle w:val="AfterTable"/></w:pPr></w:p>'
    }
    $script:quoteBuf.Clear()
}

function Flush-List {
    if ($script:listBuf.Count -eq 0) { return }
    $i = 1
    foreach ($li in $script:listBuf) {
        if ($script:listType -eq 'ol') { $prefix = "$i. " } else { $prefix = [char]0x2022 + '   ' }
        Para 'ListItem' ($prefix + $li) ''
        $i++
    }
    $script:listBuf.Clear()
    $script:listType = ''
}

function Flush-All { Flush-Table; Flush-Quote; Flush-List }
